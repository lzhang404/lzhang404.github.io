Multi-character tokens (critical)

Project 1 was mostly “single-char tokens”. Now you need lookahead.

function peek(input: string, i: number): string | null {
  return i + 1 < input.length ? input[i + 1] : null;
}

// Examples:
if (ch === "=" && peek(input, i) === "=") { /* emit == */ }
else if (ch === "=") { /* emit = */ }

if (ch === "&" && peek(input, i) === "&") { /* emit && */ }
else if (ch === "&") { throw new Error("single & is not allowed"); }

  • Same idea for !=, <=, >=, ||

Keywords vs IDENT

Lex identifiers as strings first, then re-label if it matches a keyword.

const keywords: Record<string, TokenType> = {
  let: TokenType.LET,
  if: TokenType.IF,
  else: TokenType.ELSE,
  while: TokenType.WHILE,
  return: TokenType.RETURN,
  true: TokenType.TRUE,
  false: TokenType.FALSE,
};

if (keywords[ident] !== undefined) emit(keywords[ident], ident);
else emit(TokenType.IDENT, ident);

Step 2: Extend expression precedence

Project 1 precedence ended at + and -.

Project 2 adds logical + comparison layers:

Expr → Or

Or          → And ("||" And)*
And         → Equality ("&&" Equality)*
Equality    → Comparison (("==" | "!=") Comparison)*
Comparison  → Add (("<" | "<=" | ">" | ">=") Add)*
Add         → Mul (("+"|"-") Mul)*
Mul         → Power (("*"|"/") Power)*
Power       → Unary ("^" Power)?        // right-associative
Unary       → ("+"|"-"|"!") Unary | Primary
Primary     → NUM | true | false | IDENT | "(" Expr ")"

Parser shape (same pattern as Project 1)

You already used “one function per precedence level”. Keep that style.

// New top:
parseExpr(): ExprNode { return this.parseOr(); }

parseOr(): ExprNode {
  let node = this.parseAnd();
  while (this.match(TokenType.OROR)) {
    const right = this.parseAnd();
    node = { type: "BinaryExpr", op: "||", left: node, right };
  }
  return node;
}

  • Repeat the pattern for And / Equality / Comparison

Step 3: Add statements + blocks

Program   ::= Stmt* EOF

Step 5: Scoped environments (blocks)

Project 1 used a plain object environment.

Project 2 needs nested scopes with lookup and assignment update rules.

class Env {
  constructor(public parent: Env | null,
              public values = new Map<string, Value>()) {}

  define(name: string, v: Value) { this.values.set(name, v); }

  lookup(name: string): Value {
    if (this.values.has(name)) return this.values.get(name)!;
    if (this.parent) return this.parent.lookup(name);
    throw new Error(`Undefined variable '${name}'`);
  }

  assign(name: string, v: Value): void {
    if (this.values.has(name)) { this.values.set(name, v); return; }
    if (this.parent) { this.parent.assign(name, v); return; }
    throw new Error(`Assign to undefined variable '${name}'`);
  }
}

if / while rules you must enforce

  • Condition must evaluate to a boolean, otherwise error.

  • Blocks run in a new child scope.

function asBool(v: Value): boolean {
  if (v.kind !== "boolean") throw new Error("Condition must be boolean");
  return v.value;
}

// while:
while (asBool(evalExpr(cond, env))) {
  const r = execBlock(body, new Env(env));
  if (r.type === "return") return r;
}

Short-circuit semantics (Module 6 focus)

Do not evaluate the right side if the result is already determined.

case "BinaryExpr": {
  if (node.op === "&&") {
    const left = evalExpr(node.left, env);
    const b = asBool(left);
    if (!b) return { kind: "boolean", value: false }; // skip RHS
    return evalExpr(node.right, env);                 // RHS decides
  }
  if (node.op === "||") {
    const left = evalExpr(node.left, env);
    const b = asBool(left);
    if (b) return { kind: "boolean", value: true };   // skip RHS
    return evalExpr(node.right, env);
  }
  // other operators evaluate both sides
}

Program result rule (for deterministic grading)

  • The program result is the value of the last executed expression statement.

  • If no expression statement executes, the result is null.

  • A bare return; returns null.

let x = 1;
x + 2;
Result: 3

let x = 1;
let y = 2;
Result: null

Module 7

Subprograms, Parameters, and Closures

Milestone B: finish Project 2 with functions + closures

What changes when we add functions?

  • Expressions produce more kinds of values (function values)

  • We add new AST nodes: FunExpr and CallExpr

  • We need a call boundary:

    • parameter binding

    • arity checks

    • return behavior

    • closure capture (lexical scoping)

Module 7 grammar additions

Extend Primary and add Call:

Unary       ::= ("+"|"-"|"!") Unary | Call
Call        ::= Primary ("(" ArgList? ")")*
ArgList     ::= Expr ("," Expr)*

Primary     ::= NUM | true | false | IDENT | FunExpr | "(" Expr ")"

FunExpr     ::= "fun" "(" Params? ")" Block
Params      ::= IDENT ("," IDENT)*

  • Calls bind tighter than binary operators (so f(1)+2 parses correctly)

  • ("(" ... ")") repeats so you can do makeAdder(10)(3)

New AST nodes for Module 7

interface FunExprNode {
  type: "FunExpr";
  params: string[];
  body: BlockStmt;
}

interface CallExprNode {
  type: "CallExpr";
  callee: ExprNode;
  args: ExprNode[];
}

  • body is a BlockStmt (statements, not an expression)

Example implementation: parse FunExpr

// In Primary:
if (this.match(TokenType.FUN)) {
  this.expect(TokenType.LPAREN, "Expected '(' after fun");
  const params: string[] = [];
  if (!this.check(TokenType.RPAREN)) {
    do {
      const id = this.expect(TokenType.IDENT, "Expected parameter name");
      params.push(id.lexeme);
    } while (this.match(TokenType.COMMA));
  }
  this.expect(TokenType.RPAREN, "Expected ')'");
  const body = this.parseBlock(); // expects '{'
  return { type: "FunExpr", params, body };
}

Example implementation: parse CallExpr

Calls are “postfix” after Primary and can repeat.

private parseCall(): ExprNode {
  let node = this.parsePrimary();

  while (this.match(TokenType.LPAREN)) {
    const args: ExprNode[] = [];
    if (!this.check(TokenType.RPAREN)) {
      do {
        args.push(this.parseExpr());
      } while (this.match(TokenType.COMMA));
    }
    this.expect(TokenType.RPAREN, "Expected ')'");
    node = { type: "CallExpr", callee: node, args };
  }

  return node;
}

// Then Unary becomes:
private parseUnary(): ExprNode {
  const tok = this.current();
  if (tok.type === TokenType.PLUS || tok.type === TokenType.MINUS || tok.type === TokenType.BANG) {
    this.pos++;
    const expr = this.parseUnary();
    return { type: "UnaryExpr", op: tok.lexeme, expr };
  }
  return this.parseCall();
}

Runtime values: add function values (closures)

A function value must store code + captured environment.

type Value =
  | { kind: "number"; value: number }
  | { kind: "boolean"; value: boolean }
  | { kind: "null" }
  | { kind: "function"; params: string[]; body: BlockStmt; closure: Env };

  • closure = the Env at function creation time

Example implementation: evaluating FunExpr

case "FunExpr":
  return {
    kind: "function",
    params: node.params,
    body: node.body,
    closure: env, // capture lexical environment
  };

  • If closure capture is wrong, nested functions won't “remember” variables

Call semantics (the function contract)

  • Evaluate callee (must be function)

  • Evaluate args left-to-right

  • Arity must match (else error)

  • Create callEnv whose parent is fn.closure

  • Bind params using define (like let)

  • Execute body; return result or null

Example implementation: evaluating CallExpr

case "CallExpr": {
  const calleeV = evalExpr(node.callee, env);
  if (calleeV.kind !== "function") throw new Error("Calling a non-function");

  const fn = calleeV;
  if (node.args.length !== fn.params.length) throw new Error("Arity mismatch");

  // Evaluate args left-to-right
  const argVals = node.args.map(a => evalExpr(a, env));

  // New call environment chains to closure
  const callEnv = new Env(fn.closure);

  for (let i = 0; i < fn.params.length; i++) {
    callEnv.define(fn.params[i], argVals[i]);
  }

  const r = execBlock(fn.body, callEnv);
  if (r.type === "return") return r.value;
  return { kind: "null" };
}

Return bubbling (why ExecResult matters)

Return has to escape nested blocks/loops up to the call site.

function execBlock(b: BlockStmt, env: Env): ExecResult {
  const blockEnv = new Env(env); // block scope
  for (const s of b.stmts) {
    const r = execStmt(s, blockEnv);
    if (r.type === "return") return r; // bubble up
  }
  return { type: "normal" };
}
